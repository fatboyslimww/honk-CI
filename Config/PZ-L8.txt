# CMCC PZ-L8 (IPQ5018 + QCA8337 + QCN6122)
#
# Dedicated NSS build: OpenWrt 25.12.5 + mufeng05/openwrt-cmcc-pz-l8-nss.
# The board files, QSDK NSS patches and the 256 MB pbuf/overlay tuning come
# from that project (pinned by Scripts/apply-pz-l8.sh).
#
# Deliberately minimal: NSS forwarding + WiFi + honk/dae + zram + luci-light.
# No DPD/eBPF, no smartdns, no homeproxy/passwall/openclash, no Docker.
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq50xx=y
CONFIG_TARGET_qualcommax_ipq50xx_DEVICE_cmcc_pz-l8=y

# --- Qualcomm NSS / QSDK offload ------------------------------------------

# (512 MB) because that is the combination measured stable on this 256 MB
# board; the 256 MB-specific pbuf tuning lives in the reference rootfs overlay.
CONFIG_NSS_FIRMWARE_VERSION_12_5=y
CONFIG_NSS_MEM_PROFILE_MEDIUM=y
CONFIG_NSS_DRV_IPV6_ENABLE=y
CONFIG_NSS_DRV_LSO_RX_ENABLE=y

# --- honk / dae -------------------------------------------------------------

CONFIG_PACKAGE_luci-app-honk=y
CONFIG_PACKAGE_v2ray-geoip=y
CONFIG_PACKAGE_v2ray-geosite=y
