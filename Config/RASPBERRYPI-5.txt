#设备平台：树莓派 5/500/CM5（bcm2712, Cortex-A76, 64bit）
CONFIG_TARGET_bcm27xx=y
CONFIG_TARGET_bcm27xx_bcm2712=y
#设备列表
CONFIG_TARGET_DEVICE_bcm27xx_bcm2712_DEVICE_rpi-5=y
#USB 网卡（树莓派只有一个板载网口，做路由一般外接 USB 网卡作 WAN/LAN）
CONFIG_PACKAGE_kmod-usb-net-rtl8152=y
CONFIG_PACKAGE_kmod-usb-net-asix-ax88179=y
CONFIG_PACKAGE_r8152-firmware=y
#WIFI：板载 BCM43455/43430 由设备默认包（brcmfmac + 固件）提供
CONFIG_PACKAGE_wpad-openssl=y
#风扇/温控（设备默认包已带，显式写出防止被裁）
CONFIG_PACKAGE_kmod-hwmon-pwmfan=y
CONFIG_PACKAGE_kmod-thermal=y
#组件调整
CONFIG_TARGET_KERNEL_PARTSIZE=128
CONFIG_TARGET_ROOTFS_PARTSIZE=1024
CONFIG_TARGET_ROOTFS_EXT4FS=n
